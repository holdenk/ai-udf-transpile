```python
def collatz_steps(n: int) -> int:
    if n is None or n < 1:
        return None
    steps = 0
    while n != 1:
        n = n // 2 if n % 2 == 0 else 3 * n + 1
        steps += 1
    return steps
```

```java
package ai_udf;

import java.math.BigInteger;
import org.apache.spark.sql.api.java.UDF1;

public class CollatzSteps implements UDF1<Object, Object> {
    private static final long LIMIT = (Long.MAX_VALUE - 1) / 3;

    @Override
    public Object call(Object arg0) {
        if (arg0 == null) return null;
        long n = ((Number) arg0).longValue();
        if (n < 1) return null;
        long steps = 0;
        while (n != 1) {
            if ((n & 1) == 0) {
                n >>= 1;
            } else if (n > LIMIT) {
                return big(BigInteger.valueOf(n), steps);
            } else {
                n = 3 * n + 1;
            }
            steps++;
        }
        return steps;
    }

    private static Object big(BigInteger n, long steps) {
        BigInteger three = BigInteger.valueOf(3);
        while (!n.equals(BigInteger.ONE)) {
            n = n.testBit(0) ? n.multiply(three).add(BigInteger.ONE) : n.shiftRight(1);
            steps++;
        }
        return steps;
    }
}
```

- **Rust (ensemble / coco/auto)**: rust UDF (Comet native): (attempt_count=1, hypothesis_passed=True) — accepted — 2.9M rows/s (3.3x vs python, Comet native)

```rust
pub fn udf(n: Option<i64>) -> Result<Option<i64>, String> {
    // i128: 3n + 1 can leave i64 on the way to a result that fits.
    let mut n = match n {
        Some(v) if v >= 1 => v as i128,
        _ => return Ok(None),
    };
    let mut steps: i64 = 0;
    while n != 1 {
        n = if n % 2 == 0 {
            n / 2
        } else {
            n.checked_mul(3)
                .and_then(|m| m.checked_add(1))
                .ok_or_else(|| "collatz intermediate overflowed i128".to_string())?
        };
        steps += 1;
    }
    Ok(Some(steps))
}
```

```rust
fn fold(c: char) -> char {
    match c {
        '\u{130}' | '\u{131}' => 'i',
        '\u{17f}' => 's',
        '\u{212a}' => 'k',
        _ => c.to_ascii_lowercase(),
    }
}

fn search(text: &[char], lit: &str) -> Option<String> {
    let lit: Vec<char> = lit.chars().collect();
    let n = lit.len();
    if text.len() <= n {
        return None;
    }
    for i in 0..(text.len() - n) {
        if (0..n).all(|j| fold(text[i + j]) == lit[j]) && text[i + n] != '.' {
            return Some(text[i + n..].iter().take_while(|&&c| c != '.').collect());
        }
    }
    None
}
```

```rust
fn eqci(c: char, lit: char) -> bool {
    if c == lit {
        return true;
    }
    let mut it = c.to_lowercase();
    if it.next() == Some(lit) && it.next().is_none() {

    }

fn sub_delim(s: &[char], open: &[char], close: &[char]) -> Vec<char> {
    let n = s.len();
    let mut out = Vec::with_capacity(n);
    let mut i = 0;
    while i < n {
        if starts(s, i, open) {

            let mut k = start;

            }

fn sub_phrase(s: &[char], words: &[&str]) -> Vec<char> {
    let n = s.len();
    let mut out = Vec::with_capacity(n);
    let mut i = 0;
    while i < n {
        let mut eol = i;
        while eol < n && s[eol] != '\n' {
            eol += 1;

        let mut p = eol;

                break;
```

```rust
fn is_py_space(c: char) -> bool {
    matches!(c,
        '\u{09}'..='\u{0d}' | '\u{1c}'..='\u{20}' | '\u{85}' | '\u{a0}' | '\u{1680}'
        | '\u{2000}'..='\u{200a}' | '\u{2028}' | '\u{2029}' | '\u{202f}' | '\u{205f}' | '\u{3000}')
}

// Python's simple (one code point) lowercase; U+0130 is the only full mapping that expands.
fn simple_lower(c: char) -> char {
    if c == '\u{130}' {
        return 'i';
    }
    let mut it = c.to_lowercase();
    match (it.next(), it.next()) {
        (Some(l), None) => l,
        _ => c,
    }
}

fn lit_matches(l: char, ch: char) -> bool {
    if !l.is_ascii_lowercase() {
        return ch == l;
    }
    // sre's extra ignorecase equivalences: i/dotless i, s/long s.
    simple_lower(ch) == l || (l == 'i' && ch == '\u{131}') || (l == 's' && ch == '\u{17f}')
}

enum Job {
    Run(usize, usize),
    Restore(usize, usize),
}

fn search_prog(prog: &[Inst], text: &[char], visited: &mut Vec<u64>) -> Option<(usize, usize)> {
    let n = text.len();
    let width = n + 1;
    let bits = prog.len() * width;
    visited.clear();
    visited.resize(bits / 64 + 1, 0);
    let mut caps = [0usize; 2];
    let mut stack: Vec<Job> = Vec::with_capacity(64);
    // A (pc, sp) that failed once fails again: no backreferences, so share visited across starts.
    for start in 0..=n {
        stack.push(Job::Run(0, start));
        while let Some(job) = stack.pop() {
            let (mut pc, mut sp) = match job {
                Job::Run(pc, sp) => (pc, sp),
                Job::Restore(k, v) => {
                    caps[k] = v;
                    continue;
                }
            };
            loop {
                let idx = pc * width + sp;
                if visited[idx / 64] & (1u64 << (idx % 64)) != 0 {
                    break;
                }
                visited[idx / 64] |= 1u64 << (idx % 64);
                let ok = match prog[pc] {
                    Inst::Lit(l) => sp < n && lit_matches(l, text[sp]),
                    Inst::Space => sp < n && is_py_space(text[sp]),
                    Inst::Word => sp < n && (text[sp].is_alphanumeric() || text[sp] == '_'),
                    Inst::NotDot => sp < n && text[sp] != '.',
                    Inst::Any => sp < n && text[sp] != '\n',
                    Inst::Split(a, b) => {
                        stack.push(Job::Run(b, sp));
                        pc = a;
                        continue;
                    }
                    Inst::Jmp(a) => {
                        pc = a;
                        continue;
                    }
                    Inst::Save(k) => {
                        stack.push(Job::Restore(k, caps[k]));
                        caps[k] = sp;
                        pc += 1;
                        continue;
                    }
                    Inst::Match => return Some((caps[0], caps[1])),
                };
                if !ok {
                    break;
                }
                pc += 1;
                sp += 1;
            }
        }
    }
    None
}

pub fn udf(findings: Option<&str>, tc: Option<&str>, tsc: Option<&str>) -> Result<Option<String>, String> {
    if let Some(f) = findings {
        if !f.is_empty() {
            let text: Vec<char> = f.chars().collect();
            let mut visited: Vec<u64> = Vec::with_capacity(1024);
            for (i, prog) in build_patterns().iter().enumerate() {
                if let Some((s, e)) = search_prog(prog, &text, &mut visited) {
                    if i == 8 {
                        return Ok(None);
                    }
                    if e - s > 2 {
                        return Ok(Some(text[s..e].iter().collect()));
                    }
                }
            }
        }
    }
    let tc = tc.unwrap_or("");
    if tc == "Other" {
        return Ok(None);
    }
    let tsc = match tsc {
        Some("Other") | None => "",
        Some(s) => s,
    };
    if !tsc.is_empty() && !tc.is_empty() {
        return Ok(Some(format!("{}{}", tc, tsc)));
    }
    Ok(None)
}
```

```rust
                    if (0xd800..0xdc00).contains(&u) {
                        let save = self.i;
                        if self.lit("\\u") {
                            if let Ok(lo) = self.hex4() {
                                if (0xdc00..0xe000).contains(&lo) {
                                    let cp = 0x10000 + ((u - 0xd800) << 10) + (lo - 0xdc00);
                                    out.push(char::from_u32(cp).unwrap_or('\u{fffd}'));
                                    continue;
                                }
                            }
                        }
                        self.i = save;
                        lone = true;
                    } else if (0xdc00..0xe000).contains(&u) {
                        lone = true;
                    } else {
                        out.push(char::from_u32(u).unwrap_or('\u{fffd}'));
                    }
                }
                _ => return Err("Invalid \\escape".to_string()),
            }
        }
        Ok(if lone { None } else { Some(out) })
    }

    fn digits(&mut self) -> usize {
        let s = self.i;
        while matches!(self.peek(), Some('0'..='9')) {
            self.i += 1;
        }
        self.i - s
    }

    fn number(&mut self) -> Result<Val, String> {
        let start = self.i;
        let neg = self.peek() == Some('-');
        if neg {
            self.i += 1;
        }
        match self.peek() {
            Some('0') => self.i += 1,
            Some('1'..='9') => {
                self.digits();
            }
            _ => return Err("Expecting value".to_string()),
        }
        let int_end = self.i;
        let mut is_float = false;
        if self.peek() == Some('.') && matches!(self.c.get(self.i + 1), Some('0'..='9')) {
            self.i += 1;
            self.digits();
            is_float = true;
        }
        if matches!(self.peek(), Some('e') | Some('E')) {
            let save = self.i;
            self.i += 1;
            if matches!(self.peek(), Some('+') | Some('-')) {
                self.i += 1;
            }
            if self.digits() == 0 {
                self.i = save;
            } else {
                is_float = true;
            }
        }
        let text: String = self.c[start..self.i].iter().collect();
        if is_float {
            return text
                .parse::<f64>()
                .map(Val::Float)
                .map_err(|e| e.to_string());
        }
        let mag: String = self.c[if neg { start + 1 } else { start }..int_end]
            .iter()
            .collect();
        // Python's int() digit limit.
        if mag.len() > 4300 {
            return Err("Exceeds the limit for integer string conversion".to_string());
        }
        Ok(Val::Int(if neg && mag != "0" { text } else { mag }))
    }

    fn scalar(&mut self) -> Result<Val, String> {
        match self.peek() {
            Some('"') => {
                self.i += 1;
                Ok(Val::Str(self.string()?))
            }
            Some('n') if self.lit("null") => Ok(Val::Null),
            Some('t') if self.lit("true") => Ok(Val::Bool(true)),
            Some('f') if self.lit("false") => Ok(Val::Bool(false)),
            Some('N') if self.lit("NaN") => Ok(Val::Float(f64::NAN)),
            Some('I') if self.lit("Infinity") => Ok(Val::Float(f64::INFINITY)),
            Some('-') if self.lit("-Infinity") => Ok(Val::Float(f64::NEG_INFINITY)),
            Some('-') | Some('0'..='9') => self.number(),
            _ => Err("Expecting value".to_string()),
        }
    }
}

// java.lang.Double.toString, which is what Spark prints for a Python float.
fn java_double(x: f64) -> String {
    if x.is_nan() {
        return "NaN".to_string();
    }
    if x.is_infinite() {
        return if x > 0.0 { "Infinity" } else { "-Infinity" }.to_string();
    }
    if x == 0.0 {
        return if x.is_sign_negative() { "-0.0" } else { "0.0" }.to_string();
    }
    let sign = if x < 0.0 { "-" } else { "" };
    let a = x.abs();
    let sci = format!("{:e}", a);
    let (mant, exp) = match sci.split_once('e') {
        Some((m, e)) => (m.to_string(), e.parse::<i32>().unwrap_or(0)),
        None => (sci.clone(), 0),
    };
    let digits: String = mant.chars().filter(|c| *c != '.').collect();
    if (1e-3..1e7).contains(&a) {
        let (ip, fp) = if exp >= 0 {
            let k = (exp + 1) as usize;
            if digits.len() > k {
                (digits[..k].to_string(), digits[k..].to_string())
            } else {
                (format!("{}{}", digits, "0".repeat(k - digits.len())), String::new())
            }
        } else {
            ("0".to_string(), format!("{}{}", "0".repeat((-exp - 1) as usize), digits))
        };
        let fp = if fp.is_empty() { "0".to_string() } else { fp };
        format!("{}{}.{}", sign, ip, fp)
    } else {
        let rest = if digits.len() > 1 { &digits[1..] } else { "0" };
        format!("{}{}.{}E{}", sign, &digits[..1], rest, exp)
    }
}

pub fn udf(arg0: Option<&str>) -> Result<Option<String>, String> {
    let doc = match arg0 {
        Some(d) => d,
        None => return Ok(None),
    };
    let mut p = P { c: doc.chars().collect(), i: 0 };
    p.skip_ws();
    if p.peek() != Some('{') {
        // Not JSON, or JSON without .get: Python raises either way.
        return Err("json.loads(doc) is not a dict".to_string());
    }
    let mut stack: Vec<bool> = Vec::new(); // true: object
    let mut st = St::Value;
    let mut is_widget = false;
    let mut widget: Option<Val> = None;
    loop {
        p.skip_ws();
        match st {
            St::Value | St::FirstValue => {
                if matches!(st, St::FirstValue) && p.peek() == Some(']') {
                    p.i += 1;
                    stack.pop();
                } else if p.peek() == Some('{') || p.peek() == Some('[') {
                    let obj = p.peek() == Some('{');
                    p.i += 1;
                    if is_widget {
                        widget = Some(Val::Complex);
                    }
                    is_widget = false;
                    stack.push(obj);
                    st = if obj { St::FirstKey } else { St::FirstValue };
                    continue;
                } else {
                    let v = p.scalar()?;
                    if is_widget {
                        widget = Some(v);
                    }
                    is_widget = false;
                }
            }
            St::FirstKey | St::Key => {
                if matches!(st, St::FirstKey) && p.peek() == Some('}') {
                    p.i += 1;
                    stack.pop();
                } else {
                    if p.peek() != Some('"') {
                        return Err("Expecting property name enclosed in double quotes".to_string());
                    }
                    p.i += 1;
                    let k = p.string()?;
                    p.skip_ws();
                    if p.peek() != Some(':') {
                        return Err("Expecting ':' delimiter".to_string());
                    }
                    p.i += 1;
                    is_widget = stack.len() == 1 && k.as_deref() == Some("widget");
                    st = St::Value;
                    continue;
                }
            }
            St::After => {
                let obj = *stack.last().unwrap_or(&true);
                match p.peek() {
                    Some(',') => {
                        p.i += 1;
                        st = if obj { St::Key } else { St::Value };
                        continue;
                    }
                    Some('}') if obj => {
                        p.i += 1;
                        stack.pop();
                    }
                    Some(']') if !obj => {
                        p.i += 1;
                        stack.pop();
                    }
                    _ => return Err("Expecting ',' delimiter".to_string()),
                }
            }
        }
        if stack.is_empty() {
            break;
        }
        st = St::After;
    }
    p.skip_ws();
    if p.i != p.c.len() {
        return Err("Extra data".to_string());
    }
    match widget {
        None | Some(Val::Null) => Ok(None),
        Some(Val::Str(Some(s))) => Ok(Some(s)),
        Some(Val::Str(None)) => Err("widget holds a lone surrogate".to_string()),
        Some(Val::Int(t)) => Ok(Some(t)),
        Some(Val::Bool(b)) => Ok(Some(b.to_string())),
        Some(Val::Float(f)) => Ok(Some(java_double(f))),
        Some(Val::Complex) => Err("widget is a container; its Spark string form is not reproducible".to_string()),
    }
}
```
