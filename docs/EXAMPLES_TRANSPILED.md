```java
import org.apache.spark.sql.api.java.UDF4;

public class BSPIn implements UDF4<Object, Object, Object, Object, Object> {
    public BSPIn() {}

    private static boolean contains(Object arr, Object needle) {
        scala.collection.Iterator it = ((scala.collection.Seq) arr).iterator();
        while (it.hasNext()) {
            Object e = it.next();
            if (e == null) {
                if (needle == null) return true;
            } else if (needle != null && e.equals(needle)) {
                return true;
            }
        }
        return false;
    }

    @Override
    public Object call(Object prev, Object cur, Object bsp, Object mid) {
        if (bsp == null) return null;
        if (!contains(bsp, prev)) return Integer.valueOf(0);
        if (mid == null) return null;
        return Integer.valueOf(contains(mid, cur) ? 1 : 0);
    }
}
```

```java
        if (n < 1L) {

        while (n != 1L) {
```

```rust
// Little-endian base-2^32 limbs, used only if a trajectory leaves u128.
fn big_is_one(v: &Vec<u64>) -> bool {
    v.len() == 1 && v[0] == 1
}

fn big_trim(v: &mut Vec<u64>) {
    while v.len() > 1 && *v.last().unwrap() == 0 {
        v.pop();
    }
}

fn big_half(v: &mut Vec<u64>) {
    let mut carry: u64 = 0;
    let mut i = v.len();
    while i > 0 {
        i -= 1;
        let cur = v[i] | (carry << 32);
        v[i] = cur >> 1;
        carry = cur & 1;
    }
    big_trim(v);
}

fn big_triple_plus_one(v: &mut Vec<u64>) {
    let mut carry: u64 = 1;
    for limb in v.iter_mut() {
        let cur = *limb * 3 + carry;
        *limb = cur & 0xFFFF_FFFF;
        carry = cur >> 32;
    }
    if carry > 0 {
        v.push(carry);
    }
}

fn big_from_u128(mut x: u128) -> Vec<u64> {
    let mut v = Vec::new();
    while x > 0 {
        v.push((x & 0xFFFF_FFFF) as u64);
        x >>= 32;
    }
    if v.is_empty() {
        v.push(0);
    }
    v
}

pub fn udf(arg0: Option<i64>) -> Result<Option<i64>, String> {
    let mut n: u128 = match arg0 {
        Some(v) if v >= 1 => v as u128,
        _ => return Ok(None),
    };
    let mut steps: i64 = 0;
    while n != 1 {
        if n % 2 == 0 {
            n /= 2;
        } else {
            match n.checked_mul(3).and_then(|t| t.checked_add(1)) {
                Some(t) => n = t,
                None => {
                    // Fall back to arbitrary precision, as Python ints do.
                    let mut b = big_from_u128(n);
                    while !big_is_one(&b) {
                        if b[0] % 2 == 0 {
                            big_half(&mut b);
                        } else {
                            big_triple_plus_one(&mut b);
                        }
                        steps = steps.checked_add(1).ok_or("step overflow")?;
                    }
                    return Ok(Some(steps));
                }
            }
        }
        steps += 1;
    }
    Ok(Some(steps))
}
```

```java
public class CommonRound2 implements UDF1<Object, Object> {
    public CommonRound2() {}

    @Override
    public Object call(Object arg0) {
        if (arg0 == null) {
            return null;
        }
        double x = ((Double) arg0).doubleValue();
        if (Double.isNaN(x) || Double.isInfinite(x)) {
            return Double.valueOf(x);
        }
        // Python rounds the exact binary value half-even, then converts the
        // decimal string back to the nearest double.
        java.math.BigDecimal bd = new java.math.BigDecimal(x)
```

```python
def edit_distance(a: str, b: str) -> int:
    if a is None or b is None:
        return -1
    prev = list(range(len(b) + 1))
    for i in range(1, len(a) + 1):
        cur = [i] + [0] * len(b)
        for j in range(1, len(b) + 1):
```

```python
def is_prime(n: int) -> bool:
    if n is None:
        return None
    if n < 2:
        return False
    if n % 2 == 0:

        d += 2
    return True
```

```rust
pub fn udf(arg0: Option<i64>) -> Result<Option<bool>, String> {
    let n = match arg0 {
        None => return Ok(None),
        Some(v) => v,
    };
    if n < 2 {
        return Ok(Some(false));
    }
    if n % 2 == 0 {
        return Ok(Some(n == 2));
    }
    // i128: d * d can exceed i64 near the top of the range.
    let n = n as i128;
    let mut d: i128 = 3;
    while d * d <= n {
        if n % d == 0 {
            return Ok(Some(false));
        }
```
