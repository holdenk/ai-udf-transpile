```java
package ai_udf;

import java.math.BigDecimal;
import java.math.RoundingMode;
import org.apache.spark.sql.api.java.UDF1;

public class CommonRound2 implements UDF1<Object, Object> {
    @Override
    public Object call(Object arg0) {
        if (arg0 == null) return null;
        double x = (Double) arg0;
        if (Double.isNaN(x) || Double.isInfinite(x)) return x;
        // Exact binary value, half-even: same as CPython's round(float, n).
        double r = new BigDecimal(x).setScale(2, RoundingMode.HALF_EVEN).doubleValue();
        return r == 0.0 ? Math.copySign(0.0, x) : r;
    }
}
```

```python
def edit_distance(a: str, b: str) -> int:
    if a is None or b is None:
        return -1
    prev = list(range(len(b) + 1))
    for i in range(1, len(a) + 1):
        cur = [i] + [0] * len(b)
        for j in range(1, len(b) + 1):
            cost = 0 if a[i - 1] == b[j - 1] else 1
            cur[j] = min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + cost)
        prev = cur
    return prev[len(b)]
```

```java
package ai_udf;

import org.apache.spark.sql.api.java.UDF1;

public class ExtractMonth implements UDF1<Object, Object> {
    public ExtractMonth() {}

    private static boolean pySpace(int c) {
        return (c >= 9 && c <= 13) || (c >= 0x1c && c <= 0x20) || c == 0x85 || c == 0xa0
            || c == 0x1680 || (c >= 0x2000 && c <= 0x200a) || c == 0x2028 || c == 0x2029
            || c == 0x202f || c == 0x205f || c == 0x3000;
    }

    @Override
    public Object call(Object arg0) {
        if (arg0 == null) return null;
        String s = (String) arg0;
        int f = s.indexOf('-');
        int r = s.lastIndexOf('-');
        String t;
        if (r < 0) {
            // s[0:-1]: drop the last code point
            if (s.isEmpty()) return null;
            t = s.substring(0, s.offsetByCodePoints(s.length(), -1));
        } else {
            int start = f + 1;
            if (start >= r) return null; // int('') raises
            t = s.substring(start, r);
        }
        int[] cps = t.codePoints().toArray();
        int i = 0, j = cps.length;
        while (i < j && pySpace(cps[i])) i++;
        while (j > i && pySpace(cps[j - 1])) j--;
        if (i >= j) return null;
        boolean neg = false;
        if (cps[i] == '+' || cps[i] == '-') { neg = cps[i] == '-'; i++; }
        if (i >= j) return null;
        StringBuilder digits = new StringBuilder();
        boolean prevDigit = false;
        for (int k = i; k < j; k++) {
            int c = cps[k];
            if (c == '_') {
                if (!prevDigit || k + 1 >= j) return null;
                prevDigit = false;
                continue;
            }
            if (!Character.isDigit(c)) return null;
            int d = Character.digit(c, 10);
            if (d < 0) return null;
            digits.append((char) ('0' + d));
            prevDigit = true;
        }
        if (!prevDigit) return null;
        java.math.BigInteger v = new java.math.BigInteger(digits.toString());
        if (neg) v = v.negate();
        if (v.bitLength() > 63) return null;
        return Long.valueOf(v.longValue());
    }
}
```
